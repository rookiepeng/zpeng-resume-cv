\cvsubsection{Doctoral Research at Texas Tech University}

\begin{cvskills}[4.15cm]

    \cvskill
        {3D MIMO Radar}
        {A portable K-band MIMO FMCW radar with a non-uniformly spaced 16-Tx / 16-Rx array, achieving a 90\textdegree{} field of view and $\pm$1.5\textdegree{} angular resolution in both azimuth and elevation for 3D localization mapping. (2017)}

    \cvskill
        {RF Beamforming Radar}
        {The first PCB realization of a short-range localization radar with 2D K-band beamforming, achieving $\pm$45\textdegree{} continuous beam steering using PIN-diode vector controllers, at 11.5\,$\times$\,3.9\,cm and 69\,g. (2017)}

    \cvskill
        {K-Band Multi-Mode Radar}
        {A 24-GHz portable radar configurable as FMCW or interferometry, supporting localization, range-Doppler imaging, non-contact speech sensing, wind turbine health monitoring, and UAV-borne SAR imaging. (2015--2017)}

    \cvskill
        {C-Band Multi-Mode Radar}
        {A 5.8-GHz FMCW-interferometry radar for single and multiple target tracking, vital sign detection, fall detection, range-Doppler imaging, and wind turbine structural health monitoring. (2015--2017)}

    \cvskill
        {24-GHz Radar-on-Chip}
        {A 24-GHz radar transceiver in GlobalFoundries 0.13\,\textmu m CMOS, including LC-tank VCO, poly-phase filter, RF buffers, and QFN-package matching network. (2013--2014)}

\end{cvskills}
