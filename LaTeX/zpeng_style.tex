%-------------------------------------------------------------------------------
% SHARED STYLE
%
% Single source of truth for the visual design of both zpeng_cv.tex and
% zpeng_resume.tex. Edit here, and both documents stay in sync.
%
% The design is built on one idea: a single left rail runs down every page.
% Dates, skill categories, honor years and project names all sit in that rail,
% and their content starts at the same x-position. That gives the document one
% vertical rhythm instead of four different ones, and makes it scannable by
% time (experience, education, honors) or by topic (skills, projects).
%
% The rail width is the only number that matters; everything else derives
% from it. Change \cvrailwidth and the whole document reflows coherently.
%-------------------------------------------------------------------------------

\usepackage{lastpage}
\usepackage{microtype}
\usepackage{needspace}
\usepackage{calc}

\RequirePackage[backend=biber,style=ieee,sorting=none]{biblatex}

%-------------------------------------------------------------------------------
% THE GRID
%-------------------------------------------------------------------------------
\geometry{left=1.9cm, top=1.5cm, right=1.9cm, bottom=1.6cm, footskip=.7cm}

% Sans-serif text does not want the extra sentence spacing, and abbreviations
% like "Ph.D." and "Jun." trigger it constantly in this document.
\frenchspacing

\newlength{\cvrailwidth}    \setlength{\cvrailwidth}{3.0cm}
\newlength{\cvrailgap}      \setlength{\cvrailgap}{0.45cm}

% Width of the content column, given a rail of width #1.
\newcommand{\cvbodywidth}[1]{\dimexpr\textwidth-#1-\cvrailgap\relax}

% Long unbreakable strings in bibliography entries (paper titles, URLs) can
% otherwise push a line past the right margin.
\setlength{\emergencystretch}{2.5em}

%-------------------------------------------------------------------------------
% COLOR PALETTE
%
% One accent hue (the blue used on zpeng.me, darkened for print contrast)
% plus a four-step neutral ramp. The ramp is what creates hierarchy: titles
% sit darkest, supporting metadata steps back, rules stay quiet.
%-------------------------------------------------------------------------------
\definecolor{awesome}{HTML}{1565C0}      % accent
\definecolor{accentdeep}{HTML}{0D3F73}   % deepest accent, for the surname

\definecolor{darktext}{HTML}{1A1D21}     % entry and section titles
\definecolor{text}{HTML}{2E3439}         % body copy
\definecolor{graytext}{HTML}{646E7A}     % roles, dates, secondary metadata
\definecolor{lighttext}{HTML}{959DA6}    % header location, footer
\colorlet{sectiondivider}{awesome!18!white}

% Awesome-CV tints only the first three characters of a section heading, which
% reads as a printing accident ("Sum"mary, "Hon"ors) rather than as a design.
% The whole heading carries the accent instead.
\setbool{acvSectionColorHighlight}{false}

%-------------------------------------------------------------------------------

  \renewcommand*{\footerfont}{\sffamily}
  \newcommand*{\railfont}{\sffamily\addfontfeature{Numbers=Monospaced}}

}

%-------------------------------------------------------------------------------
% THE RAIL PRIMITIVE
%
% Everything below is built on this: a right-aligned rail box, a gutter, and a
% content box. Both boxes are top-aligned so the rail label lines up with the
% first line of its content rather than floating in the middle of it.
%
% #2 must not be empty: an empty minipage collapses to zero width and the
% content silently loses its indent. Adding a \strut to guard against that
% shifts the rail's first baseline off the content's, so callers pass a label.
%-------------------------------------------------------------------------------
\long\def\cvrailline#1#2#3{%
  \noindent
  \begin{minipage}[t]{#1}\raggedleft #2\end{minipage}%
  \hspace{\cvrailgap}%
  \begin{minipage}[t]{\cvbodywidth{#1}}\raggedright #3\end{minipage}%
  \par
}

%-------------------------------------------------------------------------------
% SECTIONS
%-------------------------------------------------------------------------------
\makeatletter
% The heading sits above a hairline that spans the full measure. \needspace
% keeps a heading from being stranded alone at the foot of a page.
\renewcommand{\cvsection}[1]{%
  \needspace{5\baselineskip}%
  \par\addvspace{\acvSectionTopSkip}%
  {\sectionstyle{\MakeUppercase{#1}}}%
  \phantomsection
  \hspace{0.7em}\color{sectiondivider}\vhrulefill{1.1pt}%
  \par\addvspace{1.6mm}%
}

\renewcommand{\cvsubsection}[1]{%
  \needspace{4\baselineskip}%
  \par\addvspace{3.0mm}%
  \noindent\subsectionstyle{#1}%
  \par\addvspace{1.0mm}%
}
\makeatother

\renewcommand{\acvSectionTopSkip}{3.4mm}
\renewcommand{\acvSectionContentTopSkip}{2.0mm}

%-------------------------------------------------------------------------------
% ENTRIES (experience, education)
%
% Rendered as boxes rather than tables, so an entry is atomic: its date, role,
% organization and bullets can never be split across a page break.
%-------------------------------------------------------------------------------
\renewenvironment{cventries}{\par\addvspace{\acvSectionContentTopSkip}}{\par}

\renewcommand*{\cventry}[5]{%
  \needspace{3\baselineskip}%
  \par\addvspace{1.5mm}%
  \cvrailline{\cvrailwidth}{%
    \entrydatestyle{#4}%
  }{%
    \entrytitlestyle{#1}\par\vspace{0.5mm}%
    \notblank{#2}{\entrypositionstyle{#2}}{}%
    \notblank{#3}{\entrylocationstyle{\,\textperiodcentered\nobreak\, #3}}{}%
    \notblank{#2#3}{\par}{}%
    \vspace{0.6mm}#5%
  }%
}

% Bullets inside an entry, indented just enough to read as subordinate.
\renewenvironment{cvitems}{%
  \begin{itemize}[leftmargin=1.25em, labelwidth=0.75em, labelsep=0.5em,
                  topsep=0.5mm, partopsep=0pt, parsep=0pt, itemsep=0.4mm,
                  align=left]
    \renewcommand{\labelitemi}{\textcolor{awesome}{\raisebox{0.12ex}{\tiny$\bullet$}}}
    \cvdescfont
}{%
  \end{itemize}%
}

%-------------------------------------------------------------------------------
% LABELLED LISTS (skills, highlights, projects)
%
% The stock cvskills is a tabular* whose value column is pinned to
% 0.9\textwidth regardless of label width, so any long label pushes the value
% column off the right margin. Built on the rail primitive instead, the two
% columns always sum to the measure, and rows break across pages freely.
%
% Takes an optional rail width, for sections whose labels need more room.
%-------------------------------------------------------------------------------
\newlength{\cvskilllabelwidth}

\renewenvironment{cvskills}[1][\cvrailwidth]{%
  \setlength{\cvskilllabelwidth}{#1}%
  \par\addvspace{\acvSectionContentTopSkip}%
}{%
  \par
}

\renewcommand*{\cvskill}[2]{%
  \needspace{2\baselineskip}%
  \cvrailline{\cvskilllabelwidth}{\skilltypestyle{#1}}{\skillsetstyle{#2}}%
  \addvspace{0.9mm}%
}

%-------------------------------------------------------------------------------
% HONORS
%
% Year in the rail, award in the content column, awarding body and place as
% supporting text on the same line.
%-------------------------------------------------------------------------------
\renewenvironment{cvhonors}{%
  \par\addvspace{\acvSectionContentTopSkip}%
}{%
  \par
}

\renewcommand*{\cvhonor}[4]{%
  \needspace{2\baselineskip}%
  \cvrailline{\cvrailwidth}{\honordatestyle{#4}}{%
    \honorpositionstyle{#1}%
    \notblank{#2}{\honortitlestyle{, #2}}{}%
    \notblank{#3}{\honorlocationstyle{\,\textperiodcentered\nobreak\, #3}}{}%
  }%
  \addvspace{0.35mm}%
}

%-------------------------------------------------------------------------------
% PARAGRAPHS
%
% Indented to the content column so running text lines up with everything else.
%-------------------------------------------------------------------------------
\renewenvironment{cvparagraph}{%
  \par\addvspace{\acvSectionContentTopSkip}%
  \begin{adjustwidth}{\cvrailwidth+\cvrailgap}{0pt}%
  \paragraphstyle
}{%
  \par
  \end{adjustwidth}%
  \addvspace{1.2mm}%
}
