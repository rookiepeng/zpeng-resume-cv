% SHARED STYLE
%
% Single source of truth for the visual design of both zpeng_cv.tex and
% zpeng_resume.tex. Edit here, and both documents stay in sync.
%
% The design is built on one idea: a single left rail runs down every page.
% Dates, skill categories, honor years and project years all sit in that rail,
% and their content starts at the same x-position -- in every section, on every
% page. That gives the document one vertical rhythm instead of four different
% ones, and makes it scannable by time (experience, education, honors, projects)
% or by topic (skills, highlights).
%
% \cvrailwidth is the only number that matters; everything else derives from it.
% Change it and the whole document reflows coherently.
%-------------------------------------------------------------------------------

%-------------------------------------------------------------------------------
% THE RAIL
%
% The content column is ordinary flowing text, indented to \cvbodyindent by the
% surrounding block environment. The rail label is hung to its left with \llap,
% which occupies no horizontal space in the paragraph -- so an entry is a normal
% sequence of paragraphs and TeX may break a page in the middle of it.
%
% That is the whole point. The earlier implementation set each entry in a pair
% of minipages, which made the entry an unbreakable box: a five-bullet job
% shorter than the space left on the page was pushed whole onto the next one,
% costing up to 40% of a page every time it happened.
%
% Rail labels are expected to fit on one line. A label that wraps degrades to a
% slightly wider gap before the second line of its content (\parbox[t] carries
% the overflow as depth) -- never to an overlap -- but it should be shortened or
% \cvrailwidth widened instead.
%-------------------------------------------------------------------------------
% Rail alignment. Dates stay flush right: near-uniform digit widths hide the
% ragged edge, and the column reads as a timeline. Word labels do not, so the
% labelled lists set this to \raggedright for the length of one row.
\newcommand{\cvrailalign}{\raggedleft}

% \leavevmode is load-bearing. The label box is a \vtop underneath, and a
% \vtop takes its height from its first item only when that item is a box. Every
% rail style starts with \color, which in vertical mode contributes a whatsit
% first -- so the box gets height 0 and the label hangs one ascender, about 6pt
% here, below the line it is supposed to label. \leavevmode opens the paragraph
% before the color is set, which puts the whatsit inside the first line instead.
%
% The earlier implementation never showed this because it set the content in a
% second [t] box: both sides were displaced by the same amount and so agreed
% with each other. Against ordinary flowing text there is nothing to agree with,
% and the offset becomes visible.
\newcommand{\cvrail}[1]{%
  \llap{%
    \begin{minipage}[t]{\cvrailwidth}\leavevmode\cvrailalign #1\end{minipage}%
    \hspace{\cvrailgap}%
  }%
}

% Every block of content sits on the same axis. Built on a list rather than on
% \leftskip so that page breaks and nested itemize both behave.
% \parskip is set to half a baseline by the parskip package, which the class
% loads. Inside an entry every \par is a deliberate line break -- title, then
% organization, then bullets -- not a new paragraph, so that half-line has to go
% or the three run apart. Blocks that really do hold prose (cvparagraph) put it
% back for themselves.
\newenvironment{cvbody}{%
  \begin{adjustwidth}{\cvbodyindent}{0pt}%
  \setlength{\parskip}{0pt}%
  \raggedright
}{%
  \end{adjustwidth}%
}

\newcounter{cvsectionid}

\makeatletter
% The heading sits above a hairline that spans the full measure. \needspace
% keeps a heading from being stranded alone at the foot of a page, and
% \pdfbookmark gives the ten-page CV a usable outline in a PDF reader.
\renewcommand{\cvsection}[1]{%
  \needspace{5\baselineskip}%
  \par\addvspace{\acvSectionTopSkip}%
  \stepcounter{cvsectionid}%
  \phantomsection
  \pdfbookmark[1]{#1}{cvsec.\thecvsectionid}%
  {\sectionstyle{\MakeUppercase{#1}}}%
  \hspace{0.7em}\color{sectiondivider}\vhrulefill{1.1pt}%
  \par\addvspace{1.6mm}%
}

\renewcommand{\cvsubsection}[1]{%
  \needspace{4\baselineskip}%
  \par\addvspace{3.0mm}%
  \stepcounter{cvsectionid}%
  \phantomsection
  \pdfbookmark[2]{#1}{cvsec.\thecvsectionid}%
  \noindent\subsectionstyle{#1}%
  \par\addvspace{1.0mm}%
}
\makeatother

%-------------------------------------------------------------------------------
% PARAGRAPHS
%
% Indented to the content column so running text lines up with everything else,
% and set ragged right like every other block. Justifying a 13 cm measure of
% sans-serif full of "mm-wave" and "IEEE Trans." opens rivers between the words;
% the ragged edge is the quieter of the two flaws, and it matches the bullets.
%-------------------------------------------------------------------------------
\renewenvironment{cvparagraph}{%
  \par\addvspace{\acvSectionContentTopSkip}%
  \begin{cvbody}%
  \setlength{\parskip}{1.6mm}%
  \paragraphstyle
}{%
  \par
  \end{cvbody}%
  \addvspace{1.2mm}%
}

%-------------------------------------------------------------------------------
% BULLETED RUNS (editorial roles, committees)
%
% Short enumerations that read as a list, not as a paragraph: one item per line,

                  topsep=0pt, partopsep=0pt, parsep=0pt, itemsep=0.7mm,
                  align=left]
    \renewcommand{\labelitemi}{\textcolor{awesome}{\raisebox{0.12ex}{\tiny$\bullet$}}}%
}{%
  \end{itemize}%
  \end{cvbody}%
  \addvspace{1.2mm}%
}

%-------------------------------------------------------------------------------
% MULTI-COLUMN LISTS
%
% For long enumerations of short items -- the CV's journal-reviewer list, which
% at one item per full-measure line ate an entire page. Real columns, so they
% break across pages properly. Optional argument is the column count.
%-------------------------------------------------------------------------------
\setlength{\columnsep}{0.6cm}

\newenvironment{cvlist}[1][3]{%
  \par\addvspace{\acvSectionContentTopSkip}%
  \begin{cvbody}%
  \fontsize{8.6pt}{1.15em}\bodyfontlight\color{text}%
  \begin{multicols}{#1}%
  \raggedcolumns\raggedright
  \setlength{\parskip}{0pt}%
  \begin{itemize}[leftmargin=0.8em, labelwidth=0.5em, labelsep=0.3em,
                  topsep=0pt, partopsep=0pt, parsep=0pt, itemsep=0.35mm,
                  align=left]
    \renewcommand{\labelitemi}{\textcolor{awesome}{\raisebox{0.1ex}{\tiny$\bullet$}}}%
}{%
  \end{itemize}%
  \end{multicols}%
  \end{cvbody}%
  \addvspace{1.2mm}%
}

% \hfuzz is scoped to the header, and the \par has to be inside the group with
% it: TeX reads \hfuzz when it breaks the paragraph, not when the boxes are
% placed, so ending the paragraph outside the group would restore the old value
% first and the tolerance would do nothing.
%
% The reason any is needed: \makecvheader sets a
% zero-width photo minipage beside a \textwidth text minipage, with a source
% newline between them contributing an inter-box space. The line is therefore
% 1.7pt over the measure -- at the very end of it, so nothing is visible -- and
% reports an overfull box on every build. Tolerating it here keeps the class
% unmodified and keeps the overfull-box count in the CI summary meaningful
% instead of permanently stuck at one.
\newcommand{\makecvbanner}{%
  \begingroup
    \hfuzz=2pt\relax
    \makecvheader[L]%
    \par
  \endgroup
  \vspace{2.2mm}%
  \noindent\textcolor{awesome}{\rule{\textwidth}{1.0pt}}%
}

%-------------------------------------------------------------------------------
% PDF METADATA
%
% The class ships \hypersetup with every field blank, which leaves the reader's
% window title showing the filename and gives a recruiter's document system
% nothing to index. Called from the drivers, after \name and \position.
%
% The subject is passed in rather than taken from \@position: the position line
% carries \enskip and \cdotp for the on-page setting, and hyperref cannot turn
% those into a PDF string -- it warns and writes something mangled.
%
%   \makecvmetadata{<document kind>}{<subject>}{<keywords>}
%-------------------------------------------------------------------------------
\makeatletter
\newcommand{\makecvmetadata}[3]{%
  \hypersetup{%
    pdftitle={\@firstname\space\@lastname\space--\space#1},
    pdfauthor={\@firstname\space\@lastname},
    pdfsubject={#2},
    pdfkeywords={#3},
    pdfdisplaydoctitle=true,
  }%
}
\makeatother
